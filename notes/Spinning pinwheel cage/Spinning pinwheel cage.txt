    // rotation accumulates each beat, plus a small extra kick per beat for a "spinning up" feel
    var _rotation_per_beat = 25;

    // speed ramps up slightly each beat, so the cage feels like it's accelerating inward
    var _speed = 12 + (transition_beat_index * 2);

    for (var s = 0; s < _spoke_count; s++)
    {
        var _spoke_angle = (360 / _spoke_count) * s + _beat_rotation;

        var _spawn_x = _center_x + lengthdir_x(_radius, _spoke_angle);
        var _spawn_y = _center_y + lengthdir_y(_radius, _spoke_angle);
        var _dir     = _spoke_angle + 180; // aim back toward center

        for (var i = 0; i < _orbs_per_spoke; i++)
        {
            var _off = i * _orb_gap;
            var _ox = _spawn_x + lengthdir_x(_off, _spoke_angle); // offset further out along the same spoke line
            var _oy = _spawn_y + lengthdir_y(_off, _spoke_angle);

            with (instance_create_layer(_ox, _oy, layer, oRedOrb))
            {
                orb_fan_mode = false; // plain straight shot, no decel/fire lifecycle
