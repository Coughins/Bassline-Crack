    var _phase = 0;

    else if (_progress > 0.33) _phase = 1;

    var _base_rate, _wobble_amp, _wobble_freq, _pull_strength;
    switch (_phase)

        case 0: // slow wide swirl
            _base_rate = 0.15; _wobble_amp = 5;  _wobble_freq = 0.05;
            _pull_strength = 0.15 + 0.1 * sin(_t_rel * 0.02);
            break;
        case 1: // fast tight swirl

            _pull_strength = 0.3 + 0.2 * sin(_t_rel * 0.04);
            break;
        case 2: // chaotic mixed
            _base_rate = 0.25; _wobble_amp = 30; _wobble_freq = 0.15;
            _pull_strength = 0.25 + 0.35 * sin(_t_rel * 0.07);
            break;

    var _drift = _base_rate * _t_rel + _wobble_amp * sin(_t_rel * _wobble_freq);

    var _interval = round(lerp(6, 20, (sin(_t_rel * 0.05) * 0.5 + 0.5)));

    var _edge_patterns = [

        [0, 1, 3]     // back to top, left, right
    ];

    var _active_edges = _edge_patterns[_pattern_index];

    // local helper: spawns one orb from a given edge, orbiting the center with inward pull + drift + optional light-orb
    // _spin: +1 for clockwise, -1 for counter-clockwise
    var _spawn_orb = function(_edge, _spin, _spd, _pull, _drift)
    {
        var _sx, _sy;
        switch (_edge)
        {
            case 0: _sx = random_range(-200, 1000); _sy = -20;  break; // top
            case 1: _sx = -20;  _sy = random_range(-150, 800); break; // left
            case 3: _sx = 820;  _sy = random_range(-150, 800); break; // right
        }

        var _dir_to_center = point_direction(_sx, _sy, 400, 304);
        var _tangent_dir = _dir_to_center + (90 * _spin) + _drift;
        var _final_dir = _tangent_dir + (_pull * angle_difference(_dir_to_center, _tangent_dir));

        var _o = instance_create_layer(_sx, _sy, layer, oRedOrb_2);
        _o.direction = _final_dir;
        _o.speed = _spd;

        // idea 5: rare "light" orb — exactly 2x size, slower, glows so the player can spot it and dodge
        if (irandom(19) == 0) // ~1 in 20, tune to taste
        {
            _o.image_xscale = 2.0;
            _o.image_yscale = 2.0;
            _o.image_blend = c_white;
            _o.speed = _spd * 0.4;      // noticeably slower than the rain around it
            _o.light_radius = 140;      // this is the only kind of orb that lights up
        }

        return _o;
    }

    if (_t_rel mod _interval == 0)
    {
        for (var e = 0; e < array_length(_active_edges); e++)
        {
            var _edge = _active_edges[e];
            _spawn_orb(_edge, 1, spd, _pull_strength, _drift);  // clockwise
            _spawn_orb(_edge, -1, spd, _pull_strength, _drift); // counter-clockwise
        }
    }

    // idea 4: periodic denser pulse burst layered on top of the steady rain — bottom edge removed
    if (_t_rel mod 55 == 0)
    {
        var _burst_edges = [0, 1, 3];
        for (var e = 0; e < array_length(_burst_edges); e++)
        {
            var _edge = _burst_edges[e];
            _spawn_orb(_edge, 1, spd * 1.4, _pull_strength, _drift);
            _spawn_orb(_edge, -1, spd * 1.4, _pull_strength, _drift);
        }
    }
}
